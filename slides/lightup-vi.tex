\begin{frame}{Phát biểu bài toán tìm kiếm}
\small
\begin{tabular}{@{}p{.23\textwidth}p{.70\textwidth}@{}}
\toprule
Input & Bàn: \texttt{None} (trắng), $-1$ (tường), $0..4$ (tường số) \\

\bottomrule

\begin{frame}[fragile]{Tìm kiếm mù: Depth-First Search}
\begin{columns}[T,onlytextwidth]
\column{.60\textwidth}
\begin{lstlisting}
DFS(puzzle, limits):
  push mask 0 onto the stack
  while stack is not empty:
    if budget exhausted: return LIMIT
    state = stack.pop()
    if state in visited: continue
    if GOAL(state): return SOLVED(state)
    cell = FIRST_UNLIT_CELL(state)
    for c in bulbs illuminating cell:
      child = state | bit(c)
      if FEASIBLE(child): push child
  return UNSOLVABLE
\end{lstlisting}
\column{.36\textwidth}
\small LIFO stack + tập visited.\par\vspace{3mm}
Thứ tự rẽ nhánh cố định,\par không có thông tin heuristic.\par\vspace{3mm}
Worst case $O(2^W)$ trạng thái.\par\vspace{3mm}
Lời giải \textbf{đầu tiên} tìm được --- không tối ưu số bóng.
\end{columns}
\src{Giả mã rút gọn khớp depth\_first\_search(). TIMEOUT/NODE\_LIMIT/MEMORY\_LIMIT phân biệt UNSOLVABLE.}
\end{frame}

\begin{frame}{Heuristic của A*: ba cận dưới, lấy max}
\[
h(n)=\max\left\{\left\lceil \tfrac{D}{G_{\max}} \right\rceil,\;
\max_w d_w,\;\left\lceil \tfrac{\sum_w d_w}{H_{\max}} \right\rceil\right\}
\qquad f=g+h,\quad g=\#\text{bóng}
\]
\small $D$: số ô tối còn lại; $G_{\max}$: nhiều ô nhất một bóng hợp lệ chiếu được;\par
$d_w$: phần thiếu của tường số $w$; $H_{\max}$: nhiều kề nhất một bóng thỏa cùng lúc.
\vspace{4mm}
\begin{center}
\begin{tabular}{@{}ll@{}}
\toprule
Thành phần & Lý do là cận dưới\\
\midrule
$\lceil D/G_{\max}\rceil$ & Mỗi bóng chỉ chiếu tối đa $G_{\max}$ ô tối\\
$\max_w d_w$ & Tường thiếu $d_w$ cần ít nhất $d_w$ bóng kề\\
$\lceil \sum_w d_w/H_{\max}\rceil$ & Tổng phần thiếu cần đủ nhiều bóng kề\\
\bottomrule
\end{tabular}
\end{center}
\vspace{3mm}
Max của các cận dưới vẫn là cận dưới $\Rightarrow$ $h$ \textbf{admissible}
$\Rightarrow$ A* trả lời giải \textbf{tối ưu số bóng} khi chạy hết.
\src{Cài đặt: \_View.lower\_bound() trong algorithms/search.py.}
\end{frame}

\begin{frame}[fragile]{A* search}
\begin{columns}[T,onlytextwidth]
\column{.60\textwidth}
\begin{lstlisting}
ASTAR(puzzle, limits):
  push (f=h(0), g=0, mask 0) into min-heap
  best_g = {0: 0}
  while heap is not empty:
    if budget exhausted: return LIMIT
    pop state with minimum f
    if GOAL(state): return SOLVED(state)
    for c in bulbs illuminating
        FIRST_UNLIT_CELL(state):
      child = state | bit(c)
      if FEASIBLE(child) and
         g+1 < best_g.get(child, inf):
        best_g[child] = g+1
        push (g+1+h(child), g+1, child)
  return UNSOLVABLE
\end{lstlisting}
\column{.36\textwidth}
\small Min-heap giữ \textbf{toàn bộ frontier}.\par\vspace{3mm}
Bảng \texttt{best\_g} chống tái khám phá.\par\vspace{3mm}
Mỗi lần mở rộng tính lại ba cận dưới trên bitmask lớn.\par\vspace{3mm}
Bộ nhớ có thể tăng hàm mũ --- thấy rõ trong số liệu.
\end{columns}
\src{Giả mã rút gọn khớp astar(). Chi phí đơn vị nên f không giảm theo đường đi.}
\end{frame}

\begin{itemize}

tối ưu: 10-normal chỉ cần 9 bóng thay vì 12 (DFS thừa 33\%).

(10-hard: 151k node, $\sim$70 MiB) — TIMEOUT là \emph{không kết luận}, không
phải vô nghiệm.
\end{itemize}
