\begin{frame}{Phát biểu bài toán và đối tượng so sánh}
\begin{itemize}
\item Trạng thái: mask bóng $W$ bit, mask sáng, mask vị trí còn hợp lệ.
\item Ban đầu chưa có bóng; đích: đủ sáng, đúng tường số, không xung đột.
\item DFS: ô tối đầu tiên, thử row-major, chấp nhận rồi đi sâu ngay.
\item CSP: propagation đến điểm cố định, MRV động, ưu tiên lợi ích chiếu sáng.
\item Cùng bàn và ngân sách; khác cắt tỉa và công việc trong mỗi node.
\end{itemize}

\textbf{Cả DFS lẫn CSP không bảo đảm tối ưu số bóng.}\par
Đo hiệu ứng kết hợp MRV/CP, không tách riêng tác dụng của MRV.

\begin{frame}[fragile]{DFS: nhận bóng hợp lệ đầu rồi đi sâu}
\begin{lstlisting}
DFS(state):
  check budgets
  if GOAL(state): return SOLVED
  target = first dark cell, row-major
  for bulb in legal illuminators(target):
    if FEASIBLE(place(bulb)):
      search child immediately
      if solved or limited: return result
    exclude bulb from later siblings
  return UNSOLVABLE
\end{lstlisting}
\small Frame tường minh giữ ứng viên chưa thử; liên kết cha dựng lại đường đi.
Chỉ quay lui khi cây con thất bại. Không gian xấu nhất tăng hàm mũ.
\src{depth\_first\_search(); TIMEOUT / NODE\_LIMIT / MEMORY\_LIMIT khác UNSOLVABLE.}
\end{frame}

\begin{frame}{MRV chọn ràng buộc; CP suy ra quyết định bắt buộc}
\small
\begin{itemize}
\item Tường còn $u$ ô hợp lệ, thiếu $r$ bóng: $\binom{u}{r}$ lựa chọn.
\item Ô tối: một lựa chọn mỗi nguồn chiếu hợp lệ. Chọn miền nhỏ nhất;
hòa ưu tiên tường; thử ứng viên chiếu nhiều ô tối trước.
\item CP: tường đủ bóng $\Rightarrow$ cấm các ô kề còn lại.
\item Số bóng thiếu bằng số ô kề hợp lệ $\Rightarrow$ buộc đặt tất cả.
\item Một nguồn chiếu duy nhất $\Rightarrow$ buộc đặt bóng, loại các đường nhìn.
\item Suy luận đường chéo loại vị trí làm tường thiếu ô kề hợp lệ.
\end{itemize}
\vspace{2mm}
Lặp đến khi không đổi; có mâu thuẫn thì loại nhánh.
\src{\_View.select\_mrv\_choices() và \_View.propagate().}
\end{frame}

\begin{frame}[fragile]{Backtracking + MRV + Constraint Propagation}
\begin{lstlisting}
CSP(state):
  check budgets; propagate to fixed point
  if contradiction: reject branch
  if GOAL(state): return SOLVED
  constraint = minimum remaining choices
  for choice ordered by lighting gain:
    apply choice; propagate
    if consistent: descend immediately
    if solved or limited: return result
    restore parent checkpoint
  return UNSOLVABLE
\end{lstlisting}
\small Checkpoint phục hồi cả bóng và miền hợp lệ. Một node CSP có thể gồm
nhiều bóng bắt buộc, nên số node không đại diện lượng công việc tương đương.
\src{backtracking\_mrv\_cp(); stack tường minh tránh giới hạn đệ quy Python.}
\end{frame}
